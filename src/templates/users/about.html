{% translate "Install Floppy" %}

{% translate "Floppy is an installable PWA. Add it to your Home Screen to launch it like an app directly from your browser." %}

{% translate "iOS and iPadOS" %}

  1. {% translate "Open this instance in Safari." %}
  2. {% translate "Tap the Share button." %}
  3. {% blocktranslate %}Choose Add to Home Screen.{% endblocktranslate %}
  4. {% blocktranslate %}Turn on Open as Web App if it is shown, then tap Add.{% endblocktranslate %}

{% translate "Android" %}

  1. {% translate "Open this instance in Chrome." %}
  2. {% translate "Open the browser menu." %}
  3. {% blocktranslate %}Choose Add to Home screen.{% endblocktranslate %}
  4. {% translate "Confirm with Install, following your browser's wording." %}
